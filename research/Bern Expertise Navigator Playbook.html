  • Stop and ask the human before: pushing to any remote, making anything public, submitting, starting a paid HF endpoint, deleting anything outside mutz/.

#!/usr/bin/env python3
"""Polite snapshot crawler for mutz.

Fetches HTML only and saves raw pages to data/raw/. No LLM calls.
Extraction, embedding and search all read from this snapshot, so the
live sites are hit once and reruns cost nothing.

    pip install httpx selectolax
    export CRAWL_CONTACT="you@example.com"     # goes into the User-Agent
    python crawl.py                            # all sources
    python crawl.py artorg dcb                 # only some
    python crawl.py unibe                      # every site in inventory/unibe_seeds.txt
    python crawl.py unibe_dbmr                 # one unibe site
    python crawl.py --config my_sources.json   # override the source list

Outputs
    data/raw/<source>/<hash>.html   raw page
    data/raw/index.jsonl            one record per URL tried
    data/raw/pdf_links.txt          in-scope PDFs seen (not downloaded)
    data/raw/stats.json             per-source counts, compared on each run
"""
import argparse, asyncio, hashlib, json, os, re, sys, time
from collections import deque
from datetime import datetime, timezone
from pathlib import Path
from urllib import robotparser
from urllib.parse import urldefrag, urljoin, urlparse

DELAY = 1.0          # seconds between requests to the same host
MAX_DEPTH = 4
UA = f"mutz-hackathon-crawler/0.1 (Hack Apertus Liebefeld; contact: {os.environ.get('CRAWL_CONTACT', 'unset')})"

SKIP_EXT = re.compile(r"\.(pdf|jpe?g|png|gif|svg|webp|ico|zip|docx?|xlsx?|pptx?|mp4|mp3|css|js|xml|ics)$", re.I)

PER_DOMAIN = 4       # max requests in flight to one registrable domain (all of unibe.ch)


def expand(sources):
    """Turn a source with seeds_file into one source per listed site, so sites crawl in parallel."""
    out = []
    for src in sources:
        if "seeds_file" not in src:
            out.append(src)
            continue
        path = Path(src["seeds_file"])
        if not path.exists():
            print(f"[{src['id']}] seeds file {path} missing, skipping this source", flush=True)
            continue
        for line in path.read_text().splitlines():
            line = line.split("#", 1)[0].strip()
            if not line:
                continue
            url = clean(line if line.startswith("http") else f"https://{line}/")
            if not url:
                continue
            host = urlparse(url).netloc
            sub = host.split(":")[0].removeprefix("www.").split(".")[0]
            site = {k: v for k, v in src.items() if k != "seeds_file"}
            site.update(id=f"{src['id']}_{sub}", group=src["id"], seeds=[url], allow=[host])
            out.append(site)
    return out


def now():
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def clean(url):
    """Absolute http(s) URL without fragment or query (queries are mostly calendar traps)."""
    url, _ = urldefrag(url)
    p = urlparse(url)
    if p.scheme not in ("http", "https") or not p.netloc:
        return None
    return p._replace(query="", netloc=p.netloc.lower()).geturl()


def in_scope(url, src):
    p = urlparse(url)
    if p.netloc not in src["allow"] or SKIP_EXT.search(p.path):
        return False
    if src.get("include") and not any(re.search(rx, p.path) for rx in src["include"]):
        return False
    return not any(re.search(rx, p.path) for rx in src.get("exclude", []))


def guess_lang(url, tree):
    node = tree.css_first("html")
    lang = (node.attributes.get("lang") or "") if node else ""
    if lang:
        return lang[:2].lower()
    m = re.search(r"index_(ger|eng|fra)|/(de|en|fr)/", url)
    if m:
        return {"ger": "de", "eng": "en", "fra": "fr"}.get(m.group(1), m.group(2))
    return None


def links(tree, base):
    for a in tree.css("a[href]"):
        u = clean(urljoin(base, a.attributes.get("href") or ""))
        if u:
            yield u


class Polite:
    """Robots.txt per host, one request at a time per host, DELAY between them."""

    def __init__(self, client):
        self.client = client
        self.robots, self.locks, self.next_ok, self.delay, self.sems = {}, {}, {}, {}, {}

    def _sem(self, host):
        domain = ".".join(host.split(":")[0].split(".")[-2:])
        return self.sems.setdefault(domain, asyncio.Semaphore(PER_DOMAIN))

    async def _wait_turn(self, host):
        wait = self.next_ok.get(host, 0) - time.monotonic()
        if wait > 0:
            await asyncio.sleep(wait)

    async def allowed(self, url):
        p = urlparse(url)
        host = p.netloc
        if host not in self.robots:
            rp = robotparser.RobotFileParser()
            async with self.locks.setdefault(host, asyncio.Lock()):
                await self._wait_turn(host)
                try:
                    r = await self.client.get(f"{p.scheme}://{host}/robots.txt")
                    if r.status_code in (401, 403):
                        rp.disallow_all = True
                    else:
                        rp.parse(r.text.splitlines() if r.status_code == 200 else [])
                except httpx.HTTPError:
                    rp.parse([])
                self.next_ok[host] = time.monotonic() + DELAY
            self.robots[host] = rp
        return self.robots[host].can_fetch(UA, url)

    async def get(self, url):
        host = urlparse(url).netloc
        async with self.locks.setdefault(host, asyncio.Lock()):
            await self._wait_turn(host)
            try:
                async with self._sem(host):
                    r = await self.client.get(url)
            finally:
                self.next_ok[host] = time.monotonic() + self.delay.get(host, DELAY)
            if r.status_code in (429, 503):     # back off hard, double the delay for this host
                self.delay[host] = min(self.delay.get(host, DELAY) * 2, 30)
                retry = r.headers.get("retry-after", "")
                self.next_ok[host] = time.monotonic() + (int(retry) if retry.isdigit() else 30)
                print(f"  ! {host} said {r.status_code}, slowing to {self.delay[host]}s", flush=True)
            return r


def load_index():
    index = {}
    f = OUT / "index.jsonl"
    if f.exists():
        for line in f.read_text().splitlines():
            if line.strip():
                rec = json.loads(line)
                index[rec["url"]] = rec          # later lines win
    return index



async def crawl(src, polite, index, index_f, pdf_f):
    seeds = src["seeds"]
    group = src.get("group", src["id"])
    stats = {"fetched": 0, "cached": 0, "robots_blocked": 0, "errors": 0, "skipped": 0}
    queue, seen, pages = deque((s, 0) for s in seeds), set(), 0

    while queue and pages < src["max_pages"]:
        url, depth = queue.popleft()
        if url in seen:
            continue
        seen.add(url)
        if depth > 0 and not in_scope(url, src):
            continue

        rec = index.get(url)
        if rec and rec.get("status") == 200 and (OUT / rec["file"]).exists():
            html = (OUT / rec["file"]).read_text(errors="ignore")
            base = rec.get("final_url", url)
            stats["cached"] += 1
        else:
            if not await polite.allowed(url):
                stats["robots_blocked"] += 1
                continue
            try:
                r = await polite.get(url)
            except httpx.HTTPError as e:
                stats["errors"] += 1
                index_f.write(json.dumps({"url": url, "source": src["id"], "group": group, "status": None,
                                          "error": type(e).__name__, "fetched_at": now()}) + "\n")
                continue
            final = clean(str(r.url)) or url
            ctype = r.headers.get("content-type", "")
            if r.status_code != 200 or "html" not in ctype or urlparse(final).netloc not in src["allow"]:
                stats["skipped"] += 1
                index_f.write(json.dumps({"url": url, "final_url": final, "source": src["id"], "group": group,
                                          "status": r.status_code, "content_type": ctype,
                                          "fetched_at": now()}) + "\n")
                continue
            html = r.text
            base = final
            rel = f"{src['id']}/{hashlib.sha1(url.encode()).hexdigest()[:16]}.html"
            (OUT / rel).parent.mkdir(parents=True, exist_ok=True)
            (OUT / rel).write_text(html)
            rec = {
                "url": url, "final_url": final, "source": src["id"], "group": group, "status": 200, "file": rel,
                "fetched_at": now(), "depth": depth,
                "etag": r.headers.get("etag"), "last_modified": r.headers.get("last-modified"),
                "content_hash": hashlib.sha256(html.encode()).hexdigest(),
                "lang": guess_lang(url, HTMLParser(html)),
            }
            index[url] = rec
            index_f.write(json.dumps(rec) + "\n")
            index_f.flush()
            stats["fetched"] += 1

        pages += 1
        if pages % 25 == 0:
            print(f"[{src['id']}] {pages} pages ({stats['fetched']} new, {stats['cached']} cached), queue {len(queue)}", flush=True)
        if depth >= MAX_DEPTH:
            continue
        for link in links(HTMLParser(html), base):
            if link in seen:
                continue
            if link.lower().endswith(".pdf") and urlparse(link).netloc in src["allow"]:
                pdf_f.add(f"{src['id']}\t{link}")
                seen.add(link)
            elif in_scope(link, src):
                queue.append((link, depth + 1))

    stats["pages"] = pages
    print(f"[{src['id']}] done: {stats}", flush=True)
    return src["id"], stats


def compare_stats(new):
    f = OUT / "stats.json"
    if f.exists():
        old = json.loads(f.read_text())
        for sid, s in new.items():
            before = old.get(sid, {}).get("pages", 0)
            if before and s["pages"] < 0.7 * before:
                print(f"WARNING [{sid}] page count dropped {before} -> {s['pages']}. "
                      f"Site redesign or block? Check before trusting this crawl.", flush=True)
    f.write_text(json.dumps(new, indent=2))


async def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("only", nargs="*", help="source ids to crawl")
    ap.add_argument("--config", help="JSON file with a list of sources, replaces the built in list")
    args = ap.parse_args()

    sources = expand(json.loads(Path(args.config).read_text()) if args.config else SOURCES)
    if args.only:
        sources = [s for s in sources if s["id"] in args.only or s.get("group") in args.only]
    if "unset" in UA:
        print("Set CRAWL_CONTACT to an email so site admins can reach you instead of blocking the venue.", flush=True)

    OUT.mkdir(parents=True, exist_ok=True)
    index = load_index()
    async with httpx.AsyncClient(headers={"User-Agent": UA}, timeout=20, follow_redirects=True) as client:
        polite = Polite(client)
        pdf_file = OUT / "pdf_links.txt"
        pdfs = set(pdf_file.read_text().splitlines()) if pdf_file.exists() else set()
        with open(OUT / "index.jsonl", "a") as index_f:
            results = await asyncio.gather(*(crawl(s, polite, index, index_f, pdfs) for s in sources))
        pdf_file.write_text("".join(f"{line}\n" for line in sorted(pdfs)))
    compare_stats(dict(results))


if __name__ == "__main__":
    asyncio.run(main())

Test cases, and not overfitting to the demo

The danger with one strong demo query is that you quietly tune everything to it: a synonym here, a boost there. Then a judge types something new and it falls over. Three layers protect you.

1. Hand written cases

39 below, across 15 categories. Split into dev (tune on these) and held out (never look at them until the freeze).

2. Synthetic reverse queries

Pick random units from your crawl, have a different model write a lay company need for each, and check the unit comes back. Gives you 50+ unseen cases for free.

3. Unseen live queries

On stage, invite a judge to type their own need. If the abstain path is solid, this is the most convincing 30 seconds of the pitch.

Rules against overfitting

  • No topic words in the code. grep -ri "diabet\|glucose" src/ should return nothing. Those words belong in tests/ and the demo fixture only.
  • No hand made synonym lists. Synonyms come from MeSH and Apertus, so they work for hearing implants as well as for glucose sensors.
  • Site specifics only in sources.yaml. No if domain == "dcberne.com" in the code.
  • Tune on dev, report held out and synthetic. Run the held out set once, after the 19:00 freeze, and report the number as it is. A slightly lower honest number beats a perfect one nobody believes.
  • Report per category and per source. Show your weakest category on the slide too. Judges trust teams that know where their system is weak.
  • Stratify by unit size. Include small groups with one page, or you are only testing the big institutes.

Concepts to learn (in order)

1. Expert finding is not document search

Users want people and units, but your index holds pages and chunks. You retrieve chunks, then vote them up to the entities they mention. This is the classic expert finding problem from IR research (Balog's models).

Picture it Query "wearable glucose sensing" hits 6 chunks. 3 mention Prof A, 2 mention Lab B, 1 mentions Prof C. Score(A) = sum of the 3 chunk scores × how strongly each chunk is about A (her own profile page counts more than a news item that names her). Rank A, B, C. Each chunk is also your evidence.

Score units the same way, and let a person inherit some score from their group. That is how a small lab with one page still surfaces.

2. Hybrid retrieval with Reciprocal Rank Fusion

BM25 is great at exact jargon ("CGM", "HbA1c"). Dense embeddings are great at meaning ("sugar sensor" ≈ "glucose monitoring", across languages). Run both and merge with RRF:

score(d) = Σ over each ranker r   1 / (60 + rank_r(d))
# rank 1 in BM25 and rank 3 in dense: 1/61 + 1/63 = 0.0323
# rank 10 in only one list:            1/70        = 0.0143

No score normalization needed, which is why everyone uses it. Then rerank the top 50 with a cross encoder (bge reranker v2 m3).

3. Closing the vocabulary gap

Three layers, cheapest first:

  1. Multilingual embeddings (BGE-M3) already put "Blutzuckermessung", "glucose monitoring" and "CGM" near each other.
  2. Thesaurus expansion. MeSH has "entry terms" (synonyms) for every medical concept, is free to download, and has a German translation. Deterministic and explainable: "matched because CGM is a synonym of Continuous Glucose Monitoring (MeSH)".
  3. Apertus query rewriting. Ask it for technical terms, German and French variants, and related fields. Show the expansions in the UI so the officer can untick bad ones.

Also tag each entity with normalized expertise concepts at ingest. Then lay queries and site terms meet at the same concept ID.

4. Grounded extraction with verified quotes

LLMs invent. Your rule: the model may only output a field together with a verbatim quote from the page, and your code checks the quote exists (after whitespace normalization, fuzzy ratio ≥ 0.9). Fails go to the bin. This turns "we tried not to hallucinate" into "we proved it, 0 of N facts unsupported".

5. Entity resolution

Normalize names (strip "Prof. Dr. med.", lowercase, remove accents), block by surname, then match on email (strongest), then same unit, then fuzzy name. Keep a merge log so you can show judges why two records became one.

6. Fair discoverability

Big institutes have many pages and win naive ranking. Counter it by normalizing a person's score by how many pages mention them, capping how many results one unit takes in the top 10, and measuring exposure: share of top 10 slots going to small vs large units across your gold queries. One chart, big credibility.

7. Knowing when to say "not found"

If the best reranker score is below a threshold, or fewer than two independent pages support a match, show "No strong Bern match found" instead of padding the list. The agency literally asked for this ("we cannot oversell").

The deep link trick is URL text fragments: adding #:~:text= plus the quote to a URL makes modern browsers jump to that sentence and highlight it. If the text has changed, the browser just opens the page, so nothing breaks.

from urllib.parse import quote

def enc(s):
    return quote(s, safe="").replace("-", "%2D")   # a dash is special in fragments

def deep_link(url, q):
    w = q.split()
    frag = enc(q) if len(w) <= 8 else enc(" ".join(w[:4])) + "," + enc(" ".join(w[-4:]))
    return f"{url}#:~:text={frag}"

Avoiding AI slop

The judges will see a dozen vibe coded demos today. If yours looks like the others, they assume it was built like the others. This section is the research, boiled down to rules for this product.

Why everything AI made looks the same

Models predict the most likely next token, and the most likely design is the average of every tutorial and template they were trained on. Anthropic calls this distributional convergence: with no direction, the model lands in the safe centre. That gives you Inter, a purple gradient, a centred hero and three rounded cards. One UX Collective writer calls it "the face": the same one across Lovable, v0, Bolt and Claude.

Two things make it worse. Iterating on AI output converges further, because every revision starts from the last generic result. And banning the obvious tells just pushes models to a second wave of defaults, like cream backgrounds with a serif and a terracotta accent, or near black with one acid green. So the fix is not a ban list. The fix is making real decisions that come from your subject, and Bern hands you most of them.

Copy is half of slop

The words give it away faster than the colours. This applies to your UI text, your slides, and the text Apertus writes into the pitch.

  • Banned words: unlock, elevate, seamless, cutting edge, innovative, leverage, synergy, supercharge, revolutionize, game changer, delve, at the forefront, journey, empower.
  • Banned shapes: "It's not just X, it's Y", "Ready to...?", rhetorical questions, three adjectives in a row, emoji as bullets.
  • Buttons say exactly what happens: "Find matches", "Add to pitch", "Export PDF". Never "Get started" or "Submit".
  • Errors say what broke and what to do: "The Apertus gateway did not answer. Your brief is saved. Try again in a minute."
  • No invented people. No John Doe, no Acme. Demo data is the real crawled data.

The process that actually prevents slop

  1. Write the brief before the code. Five lines in DESIGN.md: palette with roles, type with roles, the layout concept, one signature element (the rule line), one place for personality (Mutz). Paste it into every AI coding prompt.
  2. Give constraints and a reference, never "make it look nice". "Build the matches list per DESIGN.md: rows not cards, labels in the left column, content in the right, red only on Accept."
  3. Start fresh instead of polishing slop. If a component came out generic, delete it and regenerate it from the brief. Ten rounds of "make it less generic" only converge.
  4. Real content from minute one. Use crawled researchers and long German names, never placeholders. Real content breaks layouts early, which is what you want.
  5. Run a scanner before you submit. npx impeccable detect src/ flags gradient text, overused fonts, nested cards, everything centred, bounce easing and contrast problems. It is third party, so read what it flags rather than obeying it blindly.
  6. Side by side test. Put a screenshot next to berninvest.be.ch. Does it look like it belongs to the same family?
  7. 10 second test. Show a teammate one screen for 10 seconds. Ask: what is this for, and what would you click? If they hesitate, simplify.

The verifier (20 lines that win "technical rigor")

from rapidfuzz import fuzz
import re, unicodedata

def norm(s):
    s = unicodedata.normalize("NFKC", s).lower()
    return re.sub(r"\s+", " ", s).strip()

def verified(quote, page_text, min_ratio=90):
    q, t = norm(quote), norm(page_text)
    if q in t:
        return True
    return fuzz.partial_ratio(q, t) >= min_ratio
